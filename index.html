import asyncio import random import sqlite3 from aiogram import Bot, Dispatcher, F, types from aiogram.filters import CommandStart from aiogram.utils.keyboard import InlineKeyboardBuilder, ReplyKeyboardBuilder # Токен твоего бота из @BotFather TOKEN = "YOUR_BOT_TOKEN_HERE" bot = Bot(token=TOKEN) dp = Dispatcher() # --- БАЗА ДАННЫХ --- def init_db(): conn = sqlite3.connect("game.db") cursor = conn.cursor() cursor.execute(""" CREATE TABLE IF NOT EXISTS users ( user_id INTEGER PRIMARY KEY, day INTEGER DEFAULT 1, health INTEGER DEFAULT 100, hunger INTEGER DEFAULT 100, thirst INTEGER DEFAULT 100, energy INTEGER DEFAULT 100, water INTEGER DEFAULT 2, food INTEGER DEFAULT 2, scrap INTEGER DEFAULT 0 ) """) conn.commit() conn.close() def get_user(user_id): conn = sqlite3.connect("game.db") cursor = conn.cursor() cursor.execute("SELECT * FROM users WHERE user_id = ?", (user_id,)) user = cursor.fetchone() conn.close() return user def create_user(user_id): conn = sqlite3.connect("game.db") cursor = conn.cursor() cursor.execute("INSERT OR IGNORE INTO users (user_id) VALUES (?)", (user_id,)) conn.commit() conn.close() def update_user(user_id, **kwargs): conn = sqlite3.connect("game.db") cursor = conn.cursor() set_clause = ", ".join([f"{key} = {key} + ?" for key in kwargs.keys()]) values = list(kwargs.values()) + [user_id] cursor.execute(f"UPDATE users SET {set_clause} WHERE user_id = ?", values) conn.commit() conn.close() # --- КЛАВИАТУРЫ --- def main_kb(): builder = ReplyKeyboardBuilder() builder.button(text="📊 Статус") builder.button(text="🏃 Вылазка") builder.button(text="🎒 Инвентарь") builder.button(text="💤 Отдых") builder.adjust(2, 2) return builder.as_markup(resize_keyboard=True) # --- ОБРАБОТЧИКИ КОМАНД И МЕНЮ --- @dp.message(CommandStart()) async def start_cmd(message: types.Message): create_user(message.from_user.id) await message.answer( "🌲 **Добро пожаловать в Зону 0!**\n\n" "Ты проснулся в полуразрушенном бункере. Вокруг холод, а запасы на исходе. " "Твоя цель — продержаться как можно дольше.", reply_markup=main_kb(), parse_mode="Markdown" ) @dp.message(F.text == "📊 Статус") async def show_status(message: types.Message): u = get_user(message.from_user.id) if not u: return status_text = ( f"🗓 **День:** {u[1]}\n" f"-------------------\n" f"❤️ **Здоровье:** {u[2]}%\n" f"🍗 **Голод:** {u[3]}%\n" f"💧 **Жажда:** {u[4]}%\n" f"⚡ **Энергия:** {u[5]}%\n" ) await message.answer(status_text, parse_mode="Markdown") @dp.message(F.text == "🎒 Инвентарь") async def show_inventory(message: types.Message): u = get_user(message.from_user.id) kb = InlineKeyboardBuilder() if u[6] > 0: kb.button(text="💧 Выпить воду", callback_data="use_water") if u[7] > 0: kb.button(text="🍗 Съесть еду", callback_data="use_food") kb.adjust(1) inv_text = ( f"🎒 **Твои припасы:**\n\n" f"💧 Вода: {u[6]} шт.\n" f"🍗 Еда: {u[7]} шт.\n" f"⚙️ Хлам (для крафта): {u[8]} шт." ) await message.answer(inv_text, reply_markup=kb.as_markup(), parse_mode="Markdown") @dp.message(F.text == "💤 Отдых") async def rest_cmd(message: types.Message): u = get_user(message.from_user.id) if u[3] < 15 or u[4] < 15: await message.answer("❌ Ты слишком голоден или испытываешь жажду, чтобы спать!") return # Наступление нового дня update_user( message.from_user.id, day=1, energy=min(100, 40), hunger=-15, thirst=-20, health=10 if u[2] < 100 else 0 ) await message.answer("🌙 Ты поспал. Наступил новый день.\n⚡ Энергия восстановлена, но ты проголодался.") # --- МЕХАНИКА ВЫЛАЗОК --- @dp.message(F.text == "🏃 Вылазка") async def raid_start(message: types.Message): u = get_user(message.from_user.id) if u[5] < 20: await message.answer("❌ Недостаточно энергии для вылазки! Поспи или отдохни.") return update_user(message.from_user.id, energy=-20, hunger=-5, thirst=-10) # Рандомные события на вылазке events = [ ("🏪 Вы нашли заброшенный магазин. Из угла слышен шорох.", "store"), ("📦 Посреди дороги лежит разбитый ящик снабжения.", "crate"), ("🐺 На вас из кустов выбежала дикая собака!", "dog") ] event_text, event_type = random.choice(events) kb = InlineKeyboardBuilder() if event_type == "store": kb.button(text="🔍 Обыскать магазин", callback_data="raid_search") kb.button(text="🏃 Уйти", callback_data="raid_leave") elif event_type == "crate": kb.button(text="🔓 Вскрыть ящик", callback_data="raid_crate") kb.button(text="🚶 Пройти мимо", callback_data="raid_leave") elif event_type == "dog": kb.button(text="⚔️ Сразиться", callback_data="raid_fight") kb.button(text="🏃 Убежать", callback_data="raid_flee") kb.adjust(1) await message.answer(f"🔍 **Вылазка**\n\n{event_text}", reply_markup=kb.as_markup(), parse_mode="Markdown") # --- ОБРАБОТКА ДЕЙСТВИЙ (CALLBACKS) --- @dp.callback_query(F.data.startswith("use_")) async def use_item(callback: types.CallbackQuery): item = callback.data.split("_")[1] u = get_user(callback.from_user.id) if item == "water" and u[6] > 0: update_user(callback.from_user.id, water=-1, thirst=min(100 - u[4], 30)) await callback.message.edit_text("🥤 Вы выпили воду и утолили жажду.") elif item == "food" and u[7] > 0: update_user(callback.from_user.id, food=-1, hunger=min(100 - u[3], 35)) await callback.message.edit_text("🍖 Вы поели и восполнили силы.") else: await callback.answer("У вас нет этого предмета!") @dp.callback_query(F.data.startswith("raid_")) async def handle_raid_choice(callback: types.CallbackQuery): action = callback.data.split("_")[1] if action == "search": found_food = random.randint(0, 2) found_scrap = random.randint(1, 3) update_user(callback.from_user.id, food=found_food, scrap=found_scrap) await callback.message.edit_text(f"✅ Вы нашли {found_food} еды и {found_scrap} хлама!") elif action == "crate": found_water = random.randint(1, 2) update_user(callback.from_user.id, water=found_water) await callback.message.edit_text(f"📦 В ящике оказалось {found_water} шт. чистой воды!") elif action == "fight": if random.random() > 0.4: update_user(callback.from_user.id, food=1) await callback.message.edit_text("⚔️ Вы победили собаку и получили немного мяса!") else: update_user(callback.from_user.id, health=-20) await callback.message.edit_text("🩸 Собака покусала вас! Вы потеряли 20% здоровья.") elif action in ["leave", "flee"]: await callback.message.edit_text("🚶 Вы решили не рисковать и вернулись назад.") # --- ЗАПУСК --- async def main(): init_db() print("Бот запущен!") await dp.start_polling(bot) if __name__ == "__main__": asyncio.run(main())
